-- Setup: Modeling > New Parameter > Numeric range, name "Cutoff", 400 to 800, increment 5,
-- default 560 (the constrained optimum from policy/cutoff_optimizer.py). This creates a
-- disconnected 'Cutoff' table with a 'Cutoff'[Cutoff] column.

Selected Cutoff =
SELECTEDVALUE ( 'Cutoff'[Cutoff], 560 )

Approval Rate at Cutoff =
DIVIDE (
    CALCULATE (
        COUNTROWS ( mart_application ),
        mart_application[score] >= [Selected Cutoff]
    ),
    CALCULATE ( COUNTROWS ( mart_application ), REMOVEFILTERS ( mart_application ) )
)

Expected Loss at Cutoff =
CALCULATE (
    SUMX (
        mart_application,
        mart_application[pd] * mart_application[lgd] * mart_application[ead]
    ),
    mart_application[score] >= [Selected Cutoff]
)

Revenue at Cutoff =
CALCULATE (
    SUMX ( mart_application, mart_application[expected_revenue] ),
    mart_application[score] >= [Selected Cutoff]
)

Capital at Cutoff =
CALCULATE (
    SUMX ( mart_application, mart_application[capital_required] ),
    mart_application[score] >= [Selected Cutoff]
)

Risk Adjusted Return at Cutoff =
DIVIDE (
    [Revenue at Cutoff] - [Expected Loss at Cutoff],
    [Capital at Cutoff]
)
