Display folder: Time Intelligence

Default Rate PY =
CALCULATE ( [Default Rate], SAMEPERIODLASTYEAR ( dim_date[date] ) )

Default Rate YoY =
[Default Rate] - [Default Rate PY]

Exposure Rolling 12M =
CALCULATE (
    [Total Exposure],
    DATESINPERIOD ( dim_date[date], MAX ( dim_date[date] ), -12, MONTH )
)
